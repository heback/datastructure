# C5 (선택)
def build(edges, g={}):  # 간선 목록으로 이웃 목록 dict를 만든다
    for u, v in edges:
        g.setdefault(u, []).append(v)  # u가 없으면 []를 넣고 v를 덧붙인다
    return g  # 만든 dict를 돌려준다

a = build([("A", "B")])  # 간선 한 개로 build를 부른다
b = build([("C", "D")])  # 다른 간선 한 개로 build를 부른다
print(a)
print(b)
print(a is b)  # a와 b가 같은 객체인지 출력한다

A4에서 손으로 센 값을 코드로 확인한다. 함수는 결과와 함께 확인한 칸/원소 수를 돌려준다. 4개 함수 중 세 개는 완성되어 있고, has_edge_list의 빈칸 ⑤ 하나만 채운다. G1 질의 4개를 실행해 나온 값을 활동지 2부 A9의 실측 표에 옮겨 쓰고, 손 값과 다르면 어디서 틀렸는지 한 줄 쓴다.

골격

# 반환값은 (결과, 확인한 칸/원소 수)이다. 확인한 수는 코드 안에서 직접 센다.
# 4개 함수 중 has_edge_list의 빈칸 ⑤ 하나만 채우면 된다. 나머지 3개는 이미 완성되어 있다.
# 행렬 방식: u행 v열 칸 하나만 읽는다.
def has_edge_matrix(m, u, v):
    checked = 0  # 센 수의 시작값이다
    cell = m[u][v]  # u행 v열 칸의 값을 읽는다
    checked += 1
    return cell == 1, checked  # (간선이 있는가, 센 수)를 쉼표로 묶어 돌려준다

# 리스트 방식: u의 이웃을 앞에서부터 차례로 훑는다.
def has_edge_list(g, u, v):
    checked = 0
    for w in g.get(u, []):  # u의 이웃을 하나씩 w에 담는다
        checked += ____5____      # ⑤ ____5____ 자리에 알맞은 식을 쓴다
        if w == v:  # 찾는 노드이면
            return True, checked  # 찾았으니 그 자리에서 끝낸다
    return False, checked  # 끝까지 못 찾은 경우이다

# u행의 모든 칸을 훑어 1인 칸의 노드 이름을 모은다.
def neighbors_matrix(m, u):
    out = []  # 이웃을 모을 빈 리스트이다
    checked = 0
    for j in range(len(m[u])):  # u행의 칸을 0번부터 끝까지 본다
        checked += 1
        if m[u][j] == 1:  # 간선이 있는 칸이면
            out.append(NODES[j])  # 칸 번호 j를 노드 이름으로 바꿔 모은다
    return out, checked

# u의 이웃 목록을 훑으며 하나씩 모은다.
def neighbors_list(g, u):
    out = []
    checked = 0
    for w in g.get(u, []):
        checked += 1
        out.append(w)  # 이웃 w를 모은다
    return out, checked

실측 실행과 자체 테스트

# G1 질의 4개를 실행해 A4 손 표와 비교한다. 아래 출력을 활동지 A9 표에 옮겨 쓴다.
queries = [("A", "E"), ("D", "E"), ("A", "B"), ("E", "A")]  # (시작, 끝) 질의 4개이다
for a, b in queries:  # 질의를 하나씩 꺼내 a, b에 담는다
    print("간선 확인", a + "-" + b, "행렬", has_edge_matrix(m, MAP[a], MAP[b]), "리스트", has_edge_list(g, a, b))
# 이웃 질의: D의 이웃과 확인한 수를 출력한다.
print("D의 이웃 행렬", neighbors_matrix(m, MAP["D"]), "리스트", neighbors_list(g, "D"))

# 자체 테스트: 모든 (u, v) 쌍에서 확인한 수의 정의(찾으면 그 자리에서 멈추고, 없으면 목록 전체)와 비교한다.
fails = []  # 실패 이유를 모으는 빈 리스트이다
size = len(g)  # 함수를 부르기 전 g의 크기를 적어 둔다
for a in NODES:
    for b in NODES:
        row = g.get(a, [])  # a의 이웃 목록(없으면 빈 리스트)
        truth = b in row  # 실제로 간선이 있는가(True/False)
        # row.index(b)는 b가 있는 위치(0부터)이다. 앞에서부터 센 수는 위치+1이다.
        cnt = row.index(b) + 1 if truth else len(row)
        # 내 함수의 (결과, 센 수)가 기준과 같은지 비교한다. 다르면 이유를 기록한다.
        if has_edge_matrix(m, MAP[a], MAP[b]) != (truth, 1):
            fails.append("has_edge_matrix(%s,%s)" % (a, b))
        if has_edge_list(g, a, b) != (truth, cnt):
            fails.append("has_edge_list(%s,%s)" % (a, b))
for a in NODES:
    row = g.get(a, [])
    # 이웃 질의는 모든 노드에 대해 결과 목록과 센 수를 비교한다.
    if neighbors_matrix(m, MAP[a]) != ([x for x in NODES if x in row], len(NODES)):
        fails.append("neighbors_matrix(%s)" % a)
    if neighbors_list(g, a) != (list(row), len(row)):
        fails.append("neighbors_list(%s)" % a)
# 함수가 g를 바꾸지 않았는지 확인한다.
if len(g) != size:
    fails.append("함수가 g의 크기를 바꾸었다")
# 이름이 긴 노드로도 같은 정의가 성립하는지 본다.
g2 = {"서울역": ["부산항", "대구광역시"], "부산항": ["서울역"]}  # 같은 정의가 한글 이름에서도 성립하는지 본다
for a in g2:
    for b in list(g2) + ["대구광역시"]:
        row = g2[a]
        truth = b in row
        cnt = row.index(b) + 1 if truth else len(row)
        if has_edge_list(g2, a, b) != (truth, cnt):
            fails.append("has_edge_list(긴 이름 %s,%s)" % (a, b))
# 실패 이유는 앞의 8개까지만 출력한다.
for f in fails[:8]:
    print("FAIL", f)
# 실패가 하나라도 있으면 AssertionError로 멈춘다.
assert not fails, "A9 테스트 실패: %d건 (예: %s)" % (len(fails), fails[:3])
print("A9 테스트 통과")
테스트는 내 값이 “간선 확인에서 확인한 수의 정의”에 맞는지 계산식으로 비교한다. 손 표와 값이 다르면 손 표와 코드 중 어느 쪽이 틀렸는지 한 줄로 적는다.
