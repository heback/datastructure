A11 · 실행 전에 예측하기8분 · 짝

코드를 읽고 실행 전에 두 줄의 출력을 활동지 A11에 예측해 적는다. 그다음 실행해 비교한다. 같은 호출을 줄을 나누어 두 번 한다.

class Node:  # 트리의 노드를 만드는 설계도(클래스)이다
    def __init__(self, val, left=None, right=None):  # 노드를 만들 때 자동 실행된다
        self.val = val  # 노드에 저장할 값
        self.left = left  # 왼쪽 자식(없으면 None)
        self.right = right  # 오른쪽 자식(없으면 None)

def insert(root, val):          # BST 삽입 (A7)
    if root is None:  # 빈 자리(None)인가?
        return Node(val)  # 그 자리에 새 노드를 만든다
    if val < root.val:
        root.left = insert(root.left, val)
    else:
        root.right = insert(root.right, val)
    return root  # 갱신한 트리의 루트를 돌려준다

def build(values):              # 값 목록을 차례로 삽입해 BST를 만든다
    root = None  # 빈 트리에서 시작한다
    for v in values:  # 값을 앞에서부터 하나씩 꺼낸다
        root = insert(root, v)  # v를 삽입하고 루트를 갱신한다
    return root  # 완성된 트리의 루트를 돌려준다
def collect_inorder(node, visited=[]):  # 트리의 값을 모아 리스트로 돌려준다
    if node is None:  # 빈 자리이면 멈춘다
        return visited  # 모은 리스트를 돌려준다
    collect_inorder(node.left, visited)  # 왼쪽 부분 트리를 호출한다
    visited.append(node.val)  # 현재 노드의 값을 리스트에 추가한다
    collect_inorder(node.right, visited)  # 오른쪽 부분 트리를 호출한다
    return visited  # 모은 리스트를 돌려준다

t4 = build([4, 2, 6, 1, 3, 5, 7])  # 이 순서로 삽입해 만든 트리이다

print(collect_inorder(t4))  # t4에 인자 하나로 호출해 출력한다
print(collect_inorder(t4))  # 같은 호출을 한 번 더 출력한다
