# T1-a · 시작 상태: 큐 [start], visited는 비어 있다. g는 G1.
def bfs_late(g, start):
    q = deque([start])
    visited = set()
    order = []
    while q:
        u = q.popleft()
        visited.add(u)
        order.append(u)
        for v in neighbors(g, u):
            if v not in visited:
                q.append(v)
    return order

print(bfs_late(g, "A"))

OPEN4 = ["....", "....", "....", "...."]

def bfs_count(grid, start, mark_when):      # mark_when: "enqueue" 또는 "dequeue"
    enq = 1
    maxq = 1
    q = deque([start])
    seen = set()
    if mark_when == "enqueue":
        seen.add(start)
    while q:
        u = q.popleft()
        if mark_when == "dequeue":
            if u in seen:
                continue
            seen.add(u)
        for n in grid_neighbors(grid, u[0], u[1], "."):
            if n not in seen:
                if mark_when == "enqueue":
                    seen.add(n)
                q.append(n)
                enq += 1
        maxq = max(maxq, len(q))
    return enq, maxq                        # (큐에 넣은 횟수, 큐 길이의 최댓값)

print("enqueue", bfs_count(OPEN4, (0, 0), "enqueue"))
print("dequeue", bfs_count(OPEN4, (0, 0), "dequeue"))

# T2 · 시작 상태: 두 컨테이너를 n마다 새로 만든다. 시간은 환경마다 다르므로 배수(비율)만 본다.
import time

def drain_list(n):
    a = list(range(n))
    t = time.perf_counter()
    while a:
        a.pop(0)
    return time.perf_counter() - t

def drain_deque(n):
    d = deque(range(n))
    t = time.perf_counter()
    while d:
        d.popleft()
    return time.perf_counter() - t

t = {n: (drain_list(n), drain_deque(n)) for n in (20000, 200000)}
print("첫 번째:  n이 10배가 될 때 시간은", round(t[200000][0] / t[20000][0], 1), "배")
print("두 번째:  n이 10배가 될 때 시간은", round(t[200000][1] / t[20000][1], 1), "배")
print("n=200000에서 (list 시간) / (deque 시간) =", round(t[200000][0] / t[200000][1], 1))

# T4(선택) · 시작 상태: 첫 호출은 새로 시작한 상태이고, 이어지는 두 호출은 같은 프로세스에서 이어서 실행한다.
def dfs_bad(g, u, visited=set()):
    visited.add(u)
    order = [u]
    for v in neighbors(g, u):
        if v not in visited:
            order += dfs_bad(g, v, visited)
    return order

제공 코드(완성본)

# A9 제공(완성본) · 시작 상태: seen은 호출마다 새로 만든다. dirs 순서대로 이웃을 본다.
def first_path_len(grid, start, goal, dirs):
    seen = set()

    def go(r, c, steps):
        if (r, c) == goal:
            return steps
        seen.add((r, c))
        for dr, dc in dirs:
            nr, nc = r + dr, c + dc
            if 0 <= nr < len(grid) and 0 <= nc < len(grid[0]) and grid[nr][nc] != "#" and (nr, nc) not in seen:
                res = go(nr, nc, steps + 1)
                if res is not None:
                    return res
        return None

    return go(start[0], start[1], 0)       # 목표에 처음 닿을 때의 간선 수(닿지 못하면 None)
