학습 내용 · 빈칸 ①~⑧ 채우는 길잡이
  • 활동지 1부 A2의 손 규칙 1~4단계와 A3 의사코드를 코드 줄로 옮긴 것이다. bfs_order(큐)에서 손 규칙 1(시작 노드를 넣는다)은 ①②, 규칙 2(꺼낸다)는 ③, 규칙 3(방문 기록에 쓰고 이웃을 넣는다)은 ④⑤, 규칙 4(빌 때까지 반복)는 while q이다. dfs_order(재귀)에서 ⑥은 A3 의사코드의 첫 두 줄, ⑦은 if 줄, ⑧은 호출 줄이다.
  • deque는 deque([x])로 만들고, q.append(x)는 뒤에 넣고, q.popleft()는 앞에서 꺼낸다. set은 x in s로 있는지 보고 s.add(x)로 넣는다. dfs_order의 인자는 (g, 노드, visited, order)이다.
  • visited를 언제 기록할지(큐에 넣을 때인지 꺼낼 때인지)는 ②·④·⑤에 걸려 있다. 내 코드가 어느 쪽인지 읽고 활동지 2부 A6에 쓴다. 막히면 활동지 1부 A2·A3의 표를 옆에 두고 한 줄씩 대응시킨다.

# 10주차 함수(제공). 11주차는 이 함수 위에서 진행한다.
# 간선 목록을 {노드: [이웃 목록]} 형태의 인접 리스트로 바꾼다.
def build_graph(edges, directed=False):
    g = defaultdict(list)  # 키마다 빈 리스트로 시작한다
    for u, v in edges:  # 간선 (u, v)를 하나씩 꺼낸다
        g[u].append(v)  # u의 이웃 목록 끝에 v를 붙인다
        if not directed:  # 무방향이면 반대쪽도 기록한다
            g[v].append(u)
    return g

# yield: 값을 하나 내주고 멈췄다가 이어 가는 함수(제네레이터)
def neighbors(g, u):              # 없는 노드를 물어도 g가 변하지 않는 제네레이터
    if u in g:  # g에 있는 노드만 꺼낸다
        yield from g[u]  # 이웃을 차례로 하나씩 내준다

# A2에서 손으로 탐색한 친구 관계 G1(무방향). 이웃 순서는 목록 순서이다.

# A6 점검: 기대 순서는 적혀 있지 않다. 간선 목록에서 계산한 거리와 탐색 규칙으로 내 결과를 검사한다.
# 이 점검의 비교식이 무엇을 확인하는지는 A12에서 다시 이야기한다. 지금은 읽지 않고 실행만 해도 된다.
# 기준값: 탐색 규칙과 상관없이 s에서 각 노드까지의 최소 간선 수를 구한다.
def true_dist(gr, s):
    # 탐색 순서와 무관하게 s에서 각 노드까지의 최소 간선 수를 구한다(줄여 갈 수 있으면 계속 줄인다)
    d = {s: 0}  # 딕셔너리 {노드: 거리}. 시작은 0
    changed = True  # 값이 바뀌면 True로 두고 다시 훑는다
    while changed:
        changed = False
        for u in list(d):  # 훑는 중 d가 바뀌므로 복사해서 쓴다
            for v in gr.get(u, []):
                # 아직 거리를 모르는 노드는 아주 큰 값(10 ** 9)으로 본다
                if d[u] + 1 < d.get(v, 10 ** 9):
                    d[v] = d[u] + 1
                    changed = True
    return d

# 내 BFS 결과 order가 어긋난 곳을 문장 목록으로 돌려준다(빈 목록이면 통과).
# 아래 if들은 어긋난 점을 하나씩 찾는다. 걸리면 그 문장이 FAIL로 출력된다.
def bfs_problems(gr, s, order):
    d = true_dist(gr, s)
    if not isinstance(order, list):
        return ["order가 리스트가 아니다(return을 확인한다)"]
    if len(order) != len(set(order)):
        return ["같은 노드가 order에 두 번 이상 나온다"]
    if order[:1] != [s]:
        return ["order가 시작 노드로 시작하지 않는다"]
    if set(order) != set(d):
        return ["방문한 노드의 집합이 시작 노드에서 닿을 수 있는 노드와 다르다"]
    if any(d[a] > d[b] for a, b in zip(order, order[1:])):
        return ["방문 순서가 A2에서 손으로 따른 큐 규칙의 순서와 다르다"]
    # 딕셔너리 컴프리헨션: 노드마다 방문 순번을 기록한다
    pos = {x: i for i, x in enumerate(order)}
    keys = []
    for v in order[1:]:
        # v의 부모(한 단계 앞 노드 중 가장 먼저 방문된 노드)를 찾는다.
        # key=lambda u: ...는 비교 기준을 정하는 이름 없는 함수이다.
        par = min((u for u in gr.get(v, []) if d.get(u) == d[v] - 1), key=lambda u: pos[u])
        keys.append((d[v], pos[par], gr[par].index(v)))
    if any(a >= b for a, b in zip(keys, keys[1:])):
        return ["방문 순서가 A2에서 손으로 따른 큐 규칙의 순서와 다르다(이웃 목록 순서를 확인한다)"]
    return []  # 어긋난 점이 없다 = 통과

# 내 DFS 결과를 검사한다. 되돌아가는 경로 path를 따로 따라가며 확인한다.
def dfs_problems(gr, s, order):
    if not isinstance(order, list):
        return ["order가 리스트가 아니다(return을 확인한다)"]
    if len(order) != len(set(order)):
        return ["같은 노드가 order에 두 번 이상 나온다"]
    if order[:1] != [s]:
        return ["order가 시작 노드로 시작하지 않는다"]
    seen, path = {s}, [s]  # 한 줄에서 두 변수를 함께 정한다
    for x in order[1:]:
        while path and all(v in seen for v in gr.get(path[-1], [])):
            path.pop()
        if not path:
            return ["더 갈 곳이 없는데 노드가 더 방문되었다"]
        first = [v for v in gr.get(path[-1], []) if v not in seen][0]
        if x != first:
            return ["%r 다음에 %r을 방문했다. 이웃 목록에서 아직 안 본 첫 이웃을 먼저 가야 한다" % (path[-1], x)]
        seen.add(x)
        path.append(x)
    while path and all(v in seen for v in gr.get(path[-1], [])):
        path.pop()
    if path or set(order) != set(true_dist(gr, s)):
        return ["방문한 노드의 집합이 시작 노드에서 닿을 수 있는 노드와 다르다"]
    return []

# 함수를 감싸 호출 횟수를 센다. 한도를 넘으면 끝나지 않는 반복으로 보고 멈춘다.
# (무한 반복으로 컴퓨터가 멈추는 일을 막는 장치이다)
def limited(real, limit, what):
    cnt = [0]  # 리스트에 담아야 안쪽 함수가 값을 바꿀 수 있다
    def wrapped(*a):  # *a: 인자를 몇 개든 그대로 받는다
        cnt[0] += 1
        if cnt[0] > limit:
            raise RuntimeError(what + "를 너무 많이 불렀다(끝나지 않는 반복이 의심된다)")
        return real(*a)  # 원래 함수를 그대로 실행한다
    return wrapped

# 검사할 그래프 목록: (이름, 간선 목록, 시작 노드)
CASES = [
    ("G1, 시작 A", G1_EDGES, "A"),
    ("G1, 시작 D", G1_EDGES, "D"),
    ("G1, 시작 F(고립)", G1_EDGES, "F"),
    ("고리 모양", [(1, 2), (2, 3), (3, 4), (4, 1)], 1),
    ("갈래가 많은 그래프", [("x", "y"), ("x", "z"), ("y", "w"), ("z", "w"), ("w", "v"), ("v", "y")], "x"),
]
fails = []  # 실패 사유를 모아 둔다
real_neighbors = neighbors  # 원래 neighbors를 보관해 둔다
for name, edges, s in CASES:
    gr = build_graph(edges)
    # 탐색 전 그래프를 복사해 두고, 끝난 뒤 변하지 않았는지 비교한다
    keep = {k: list(v) for k, v in gr.items()}
    # neighbors에 호출 횟수 제한을 건다
    neighbors = limited(real_neighbors, 300, "neighbors")
    try:
        for fn, chk in (("bfs_order", bfs_problems), ("dfs_order", dfs_problems)):
            try:
                if fn == "bfs_order":
                    out = bfs_order(gr, s)
                else:
                    out = dfs_order(gr, s, set(), [])
            except RuntimeError as e:
                fails.append("%s %s: %s" % (name, fn, e))
                continue
            for p in chk(gr, s, out):
                fails.append("%s %s: %s" % (name, fn, p))
    # 오류가 나도 원래 neighbors로 되돌린다
    finally:
        neighbors = real_neighbors
    if {k: list(v) for k, v in gr.items()} != keep:
        fails.append("%s: 탐색하다가 그래프 g가 바뀌었다" % name)
# 실패 사유를 모두 출력한다
for f in fails:
    print("FAIL", f)
# assert: 조건이 거짓이면 AssertionError로 멈춘다.
# 실패가 하나라도 있으면 멈추므로, 위에 출력된 FAIL 줄을 읽고 고친다.
assert not fails, "A6 점검 실패: %s" % fails
print("A6 점검 통과")
호출은 bfs_order(g, "A"), dfs_order(g, "A", set(), [])이다. 시작 상태는 호출마다 새 order, 큐, visited이다.

활동을 마친 뒤 열어 확인한다 · A6 빈칸 정답
  • ① deque([start]) ② visited = {start} ③ q.popleft() ④ v not in visited ⑤ visited.add(v)와 q.append(v) ⑥ visited.add(u)와 order.append(u) ⑦ v not in visited ⑧ dfs_order(g, v, visited, order)
  • ②에서 시작 노드를 처음부터 visited에 넣고 ⑤에서 큐에 넣을 때 기록하므로 “넣을 때 기록” 방식이다. 꺼낼 때 기록하는 방식으로 짜도 되지만 ③ 아래에 방문 확인·기록 줄이 더 필요하다. 두 방식의 차이는 A8 T1에서 비교한다.
  • G1, 시작 A의 결과 값은 활동지 2부 A6에 기록한 뒤 그 쪽의 확인 상자에서 비교한다.

A7 · 격자 미로 탐색하기13분 · 짝

2차원 격자 미로 MAZE를 탐색한다. 칸 하나가 노드이고 상하좌우 이웃 칸이 간선이다. grid_neighbors는 10주차 neighbors처럼 이웃을 하나씩 내주는 제네레이터이고, shortest_path_len은 S에서 E까지의 거리(걸음 수)를 돌려주며 닿을 수 없으면 -1이다. 완성한 뒤 MAZE의 S→E 거리를 구하고 닿을 수 없는 미로 한 개와 시작=도착인 경우도 실행해 활동지 2부 A7에 기록한다.

     열 0 1 2 3 4 5 6
행 0     S . . # . . .         칸 (행, 열)이 노드이고 상하좌우로 맞닿은 길 칸이 이웃이다.
행 1     . # . # . # .         '#'은 벽이라 지나갈 수 없다. S=(0,0), E=(4,6).
행 2     . # . . . # .         이웃은 DIRS 순서(우, 하, 좌, 상)로 하나씩 내준다.
행 3     . # # # # # .
행 4     . . . . . . E
  • grid_neighbors: 방향마다 한 칸 옮긴 위치 (nr, nc)를 구해, ⑨ 격자 안인지(행은 0 이상 len(grid) 미만, 열은 0 이상 len(grid[0]) 미만)와 ⑩ passable에 든 글자인지를 모두 만족하면 ⑪ yield로 내준다. 그 칸의 글자는 grid[nr][nc]로 읽으므로 범위를 먼저 확인해야 한다(and는 앞이 거짓이면 뒤를 읽지 않는다).
  • shortest_path_len은 bfs_order와 같은 큐 틀이다. 다른 점은 visited 대신 dist(칸 → 시작에서의 걸음 수)를 쓰는 것이다. ⑫ 도착 칸의 거리를 돌려주고, ⑬ dist에 아직 없는 칸이면 ⑭ 이웃의 거리를 현재 칸 거리보다 1 크게 기록하고 큐에 넣는다. 도착 칸을 만나지 못하고 큐가 비면 -1이다.

# A7 점검: 기대 값은 적혀 있지 않다. 격자의 모든 칸을 훑는 다른 계산으로 구한 값과 비교한다.
import inspect  # 제네레이터인지 검사하는 도구

# 기준값 1: 모든 칸을 훑어 (r, c)와 상하좌우로 맞닿은 통과 가능 칸을 모은다.
# 집합 컴프리헨션이다. (행 차이 + 열 차이 = 1)이면 맞닿은 칸이다.
def ref_adj(grid, r, c, passable):
    # 모든 칸 중 (r, c)와 상하좌우로 맞닿고 통과할 수 있는 칸을 모은다(경계 검사를 따로 쓰지 않는 계산)
    return {(i, j) for i in range(len(grid)) for j in range(len(grid[0]))
            if abs(i - r) + abs(j - c) == 1 and grid[i][j] in passable}

# 기준값 2: 시작 칸에서 각 칸까지의 최소 거리를 구한다.
# 더 짧은 거리가 나오는 동안 되풀이해 훑는다.
def ref_dist(grid, start):
    d = {start: 0}  # 딕셔너리 {칸: 거리}. 시작 칸은 0
    changed = True  # 값이 바뀌면 True로 두고 다시 훑는다
    while changed:
        changed = False
        for cell in list(d):  # 훑는 중 d가 바뀌므로 복사해서 쓴다
            for n in ref_adj(grid, cell[0], cell[1], ".SE"):
                # 아직 모르는 칸은 아주 큰 값(10 ** 9)으로 본다
                if d[cell] + 1 < d.get(n, 10 ** 9):
                    d[n] = d[cell] + 1
                    changed = True
    return d

# 함수를 감싸 호출 횟수를 센다. 한도를 넘으면 끝나지 않는 반복으로 보고 멈춘다.
# (무한 반복으로 컴퓨터가 멈추는 일을 막는 장치이다)
def limited(real, limit, what):
    cnt = [0]  # 리스트에 담아야 안쪽 함수가 값을 바꿀 수 있다
    def wrapped(*a):  # *a: 인자를 몇 개든 그대로 받는다
        cnt[0] += 1
        if cnt[0] > limit:
            raise RuntimeError(what + "를 너무 많이 불렀다(끝나지 않는 반복이 의심된다)")
        return real(*a)  # 원래 함수를 그대로 실행한다
    return wrapped

# 검사할 격자들(미로, 한 줄짜리, 한 칸짜리 등)
GRIDS = [MAZE, ["S#E"], ["S.E"], ["S"], ["S..", ".#.", "..E"], ["S.#.E", "..#..", "....."], ["S#.", "#..", "..E"]]
fails = []  # 실패 사유를 모아 둔다
# yield를 쓰는 제네레이터가 아니면 실패이다
if not inspect.isgenerator(grid_neighbors(MAZE, 0, 0, ".SE")):
    fails.append("grid_neighbors가 제네레이터가 아니다(yield로 하나씩 내준다)")
for k, grid in enumerate(GRIDS):
    for r in range(len(grid)):
        for c in range(len(grid[0])):
            got = list(grid_neighbors(grid, r, c, ".SE"))  # 내 함수가 내준 이웃
            ref = ref_adj(grid, r, c, ".SE")  # 기준 계산의 이웃
            # 이웃 칸이 기준과 다르거나, 같은 칸을 중복해서 내주면 실패이다
            if set(got) != ref or len(got) != len(ref):
                fails.append("격자 %d의 (%d, %d): 이웃이 다르다" % (k, r, c))
            # 내준 이웃의 방향 번호가 오름차순이면 DIRS 순서를 지킨 것이다
            idx = [DIRS.index((a - r, b - c)) for a, b in got if (a - r, b - c) in DIRS]
            if idx != sorted(idx):
                fails.append("격자 %d의 (%d, %d): 이웃을 우, 하, 좌, 상 순서로 내주지 않았다" % (k, r, c))
real_gn = grid_neighbors  # 원래 함수를 보관해 둔다
try:
    for k, grid in enumerate(GRIDS):
        cells = [(r, c) for r in range(len(grid)) for c in range(len(grid[0])) if grid[r][c] in ".SE"]
        for s0 in cells:
            d = ref_dist(grid, s0)
            for t0 in cells:
                grid_neighbors = limited(real_gn, 200, "grid_neighbors")   # 호출마다 새로 센다
                try:
                    got = shortest_path_len(grid, s0, t0)
                except RuntimeError as e:
                    fails.append("격자 %d %s->%s: %s" % (k, s0, t0, e))
                    break
                # d.get(t0, -1): 닿을 수 없는 칸은 -1로 본다
                if got != d.get(t0, -1):
                    fails.append("격자 %d %s->%s: 거리가 다르다" % (k, s0, t0))
                    break
            if len(fails) > 12:
                break
# 오류가 나도 원래 함수로 되돌린다
finally:
    grid_neighbors = real_gn
# 실패 사유를 앞의 12개까지 출력한다
for f in fails[:12]:
    print("FAIL", f)
# 실패가 하나라도 있으면 AssertionError로 멈춘다. FAIL 줄을 읽고 고친다.
