# A9 점검: 정의대로 하나씩 세는 느린 방법과 내 결과를 비교한다.
def slow_equal(a, x):
    return sum(1 for v in a if v == x)

def slow_range(a, lo, hi):
    return sum(1 for v in a if lo <= v <= hi)

for a in (DUP, [], [5], [2, 2, 2]):
